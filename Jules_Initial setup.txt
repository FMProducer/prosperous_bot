set -euo pipefail

# 1) Изолированное окружение
python3 -m venv .venv
. .venv/bin/activate

# 2) Обновить установщик
python -m pip install --upgrade pip

# 3) Установить зависимости проекта (предпочитать бинарные колёса)
python -m pip install --prefer-binary -r requirements.txt

# 4) Установить сам пакет из корня репозитория (PEP 621; src-layout)
#    Это решает ModuleNotFoundError для 'prosperous_bot'
python -m pip install -e .

# 5) Страховка для прямого запуска pytest из корня (src-layout)
export PYTHONPATH="$PWD/src:${PYTHONPATH:-}"

# 6) Зафиксировать фактическое окружение для отладки снапшота
mkdir -p reports
python -m pip freeze > reports/pip-freeze.txt || true

# 7) Быстрая валидация окружения перед Snapshot
pytest -q --maxfail=1 --disable-warnings
